CREATE EVENT update_expired_packages
ON SCHEDULE EVERY 1 DAY
STARTS TIMESTAMP(CURRENT_DATE + INTERVAL 1 DAY, '00:00:00')
DO
    UPDATE member_packages
    SET package_status = 'Expired'
    WHERE expiry_date < CURDATE()
      AND package_status = 'Active';
